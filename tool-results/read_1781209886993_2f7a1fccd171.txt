   203→/** Extract last digit of price: 42000.50 → 0, 42001.23 → 3 */
   204→function getLastDigit(price: number): number {
   205→  return Math.floor(Math.abs(price) * 100) % 10
   206→}
   207→
   208→function computeDigitAnalysis(symbol: string, barrier: number = DEFAULT_BARRIER): DigitAnalysis | null {
   209→  const data = symbolDataMap.get(symbol)
   210→  if (!data || data.ticks.length === 0) return null
   211→
   212→  const ticks = data.ticks
   213→  const currentPrice = ticks[ticks.length - 1].price
   214→  const lastDigit = getLastDigit(currentPrice)
   215→
   216→  // Last 100 ticks (or whatever is available)
   217→  const recentTicks = ticks.slice(-ANALYSIS_WINDOW)
   218→
   219→  // Digit counts
   220→  const digitCounts = new Array(10).fill(0) as number[]
   221→  let evenCount = 0
   222→  let oddCount = 0
   223→  let overCount = 0
   224→  let underCount = 0
   225→
   226→  for (const tick of recentTicks) {
   227→    digitCounts[tick.lastDigit]++
   228→    if (tick.lastDigit % 2 === 0) {
   229→      evenCount++
   230→    } else {
   231→      oddCount++
   232→    }
   233→    if (tick.lastDigit > barrier) {
   234→      overCount++
   235→    } else {
   236→      underCount++
   237→    }
   238→  }
   239→
   240→  const total = recentTicks.length
   241→  const digitPercentages = digitCounts.map(c => total > 0 ? parseFloat(((c / total) * 100).toFixed(2)) : 0)
   242→
   243→  // Price stats
   244→  const firstPrice = recentTicks[0].price
   245→  const priceChange = parseFloat((currentPrice - firstPrice).toFixed(10))
   246→  const priceChangePercent = firstPrice !== 0 ? parseFloat(((priceChange / firstPrice) * 100).toFixed(6)) : 0
   247→
   248→  let highPrice = -Infinity
   249→  let lowPrice = Infinity
   250→  for (const tick of recentTicks) {
   251→    if (tick.price > highPrice) highPrice = tick.price
   252→    if (tick.price < lowPrice) lowPrice = tick.price
   253→  }
   254→
   255→  // Tick speed (ticks per second over last 10 seconds)
   256→  const now = Date.now()
   257→  const tenSecondsAgo = now - 10000
   258→  const recentSpeedTicks = ticks.filter(t => t.timestamp >= tenSecondsAgo)
   259→  const tickSpeed = recentSpeedTicks.length > 1
   260→    ? parseFloat((recentSpeedTicks.length / 10).toFixed(2))
   261→    : 0
   262→
   263→  // Streaks - check last consecutive even or odd
   264→  let streakType = ''
   265→  let streakCount = 0
   266→  for (let i = ticks.length - 1; i >= 0; i--) {
   267→    const d = ticks[i].lastDigit
   268→    const isEven = d % 2 === 0
   269→    if (streakCount === 0) {
   270→      streakType = isEven ? 'even' : 'odd'
   271→      streakCount = 1
   272→    } else {
   273→      const currentIsEven = isEven
   274→      if ((streakType === 'even' && currentIsEven) || (streakType === 'odd' && !currentIsEven)) {
   275→        streakCount++
   276→      } else {
   277→        break
   278→      }
   279→    }
   280→  }
   281→  const streaks = `${streakCount} consecutive ${streakType}`
   282→
   283→  // Tick history (last 25)
   284→  const tickHistory = ticks.slice(-25).map(t => ({
   285→    price: t.price,
   286→    timestamp: t.timestamp,
   287→    lastDigit: t.lastDigit,
   288→  }))
   289→
   290→  return {
   291→    lastDigit,
   292→    digitCounts,
   293→    digitPercentages,
   294→    evenCount,
   295→    oddCount,
   296→    overCount,
   297→    underCount,
   298→    streaks,
   299→    tickHistory,
   300→    priceChange,
   301→    priceChangePercent,
   302→    currentPrice: parseFloat(currentPrice.toFixed(10)),
   303→    highPrice: parseFloat(highPrice.toFixed(10)),
   304→    lowPrice: parseFloat(lowPrice.toFixed(10)),
   305→    tickSpeed,
   306→  }
   307→}

   309→function resolveSymbol(tvSymbol: string): string | null {
   310→  const mapping = SYMBOL_MAP[tvSymbol]
   311→  if (mapping) return mapping.internal
   312→
   313→  // Try direct internal symbol lookup
   314→  if (symbolDataMap.has(tvSymbol)) return tvSymbol
   315→
   316→  // Try case-insensitive lookup
   317→  const lower = tvSymbol.toLowerCase()
   318→  for (const [key, val] of Object.entries(SYMBOL_MAP)) {
   319→    if (key.toLowerCase() === lower) return val.internal
   320→    if (val.internal.toLowerCase() === lower) return val.internal
   321→  }
   322→  return null
   323→}

   325→// ============================================================
   326→// Binance WebSocket
   327→// ============================================================
   328→
   329→let binanceWs: WebSocket | null = null
   330→let binanceReconnectDelay = 1000
   331→
   332→function connectBinance() {
   333→  console.log(`[Binance] Connecting to WebSocket...`)
   334→  try {
   335→    binanceWs = new WebSocket(BINANCE_WS_URL)
   336→  } catch (e) {
   337→    console.error(`[Binance] Failed to create WebSocket:`, e)
   338→    scheduleBinanceReconnect()
   339→    return
   340→  }
   341→
   342→  binanceWs.onopen = () => {
   343→    console.log(`[Binance] Connected! Streaming ${BINANCE_STREAMS.length} symbols`)
   344→    binanceReconnectDelay = 1000
   345→  }
   346→
   347→  binanceWs.onmessage = (event) => {
   348→    try {
   349→      const raw = typeof event.data === 'string' ? event.data : ''
   350→      const msg = JSON.parse(raw)
   351→      if (!msg.stream || !msg.data) return
   352→
   353→      const symbol = msg.stream.replace('@trade', '')
   354→      const data = msg.data
   355→      const price = parseFloat(data.p)
   356→      const timestamp = data.T || Date.now()
   357→
   358→      if (isNaN(price)) return
   359→
   360→      const lastDigit = getLastDigit(price)
   361→      const tick: Tick = { price, timestamp, lastDigit }
   362→
   363→      let symbolData = symbolDataMap.get(symbol)
   364→      if (!symbolData) {
   365→        symbolData = { ticks: [], basePrice: price, source: 'binance', lastPollTimestamp: 0 }
   366→        symbolDataMap.set(symbol, symbolData)
   367→      }
   368→
   369→      symbolData.ticks.push(tick)
   370→      if (symbolData.ticks.length > MAX_TICKS) {
   371→        symbolData.ticks = symbolData.ticks.slice(-MAX_TICKS)
   372→      }
   373→      symbolData.basePrice = price
   374→
   375→      // Emit tick event via Socket.io
   376→      try {
   377→        const tvSymbol = INTERNAL_TO_TV[symbol] || symbol
   378→        const digits = computeDigitAnalysis(symbol)
   379→        io.emit('tick', {
   380→          symbol: tvSymbol,
   381→          price,
   382→          lastDigit,
   383→          timestamp,
   384→          digits,
   385→        })
   386→      } catch (emitErr) {
   387→        // Don't let socket.io errors crash the service
   388→        console.error('[Socket.io] Emit error:', emitErr)
   389→      }
   390→    } catch (e) {
   391→      // ignore parse errors
   392→    }
   393→  }
   394→
   395→  binanceWs.onclose = (event) => {
   396→    console.log(`[Binance] Disconnected (code: ${event.code}). Reconnecting in ${binanceReconnectDelay}ms...`)
   397→    scheduleBinanceReconnect()
   398→  }
   399→
   400→  binanceWs.onerror = (error) => {
   401→    console.error(`[Binance] Error:`, error.message || error)
   402→  }
   403→}
   404→
   405→function scheduleBinanceReconnect() {
   406→  setTimeout(() => {
   407→    binanceReconnectDelay = Math.min(binanceReconnectDelay * 2, 30000)
   408→    connectBinance()
   409→  }, binanceReconnectDelay)
   410→}

   416→async function pollForexPrices() {
   417→  try {
   418→    const response = await fetch('https://api.exchangerate-api.com/v4/latest/USD')
   419→    if (!response.ok) {
   420→      console.error(`[Forex] API returned ${response.status}`)
   421→      return
   422→    }
   423→    const data = await response.json()
   424→    const rates = data.rates
   425→
   426→    // Update forex pairs
   427→    const forexUpdates: Array<{ internal: string; tvSymbol: string; price: number }> = []
   428→
   429→    // EUR/USD
   430→    if (rates.EUR !== undefined) {
   431→      forexUpdates.push({ internal: 'EUR/USD', tvSymbol: 'FX:EURUSD', price: 1 / rates.EUR })
   432→    }
   433→    // GBP/USD
   434→    if (rates.GBP !== undefined) {
   435→      forexUpdates.push({ internal: 'GBP/USD', tvSymbol: 'FX:GBPUSD', price: 1 / rates.GBP })
   436→    }
   437→    // USD/JPY
   438→    if (rates.JPY !== undefined) {
   439→      forexUpdates.push({ internal: 'USD/JPY', tvSymbol: 'FX:USDJPY', price: rates.JPY })
   440→    }
   441→    // USD/CHF
   442→    if (rates.CHF !== undefined) {
   443→      forexUpdates.push({ internal: 'USD/CHF', tvSymbol: 'FX:USDCHF', price: rates.CHF })
   444→    }
   445→    // AUD/USD
   446→    if (rates.AUD !== undefined) {
   447→      forexUpdates.push({ internal: 'AUD/USD', tvSymbol: 'FX:AUDUSD', price: 1 / rates.AUD })
   448→    }
   449→    // USD/CAD
   450→    if (rates.CAD !== undefined) {
   451→      forexUpdates.push({ internal: 'USD/CAD', tvSymbol: 'FX:USDCAD', price: rates.CAD })
   452→    }
   453→    // NZD/USD
   454→    if (rates.NZD !== undefined) {
   455→      forexUpdates.push({ internal: 'NZD/USD', tvSymbol: 'FX:NZDUSD', price: 1 / rates.NZD })
   456→    }
   457→    // EUR/GBP
   458→    if (rates.EUR !== undefined && rates.GBP !== undefined) {
   459→      forexUpdates.push({ internal: 'EUR/GBP', tvSymbol: 'FX:EURGBP', price: rates.GBP / rates.EUR })
   460→    }
   461→    // EUR/JPY
   462→    if (rates.EUR !== undefined && rates.JPY !== undefined) {
   463→      forexUpdates.push({ internal: 'EUR/JPY', tvSymbol: 'FX:EURJPY', price: rates.JPY / rates.EUR })
   464→    }
   465→    // GBP/JPY
   466→    if (rates.GBP !== undefined && rates.JPY !== undefined) {
   467→      forexUpdates.push({ internal: 'GBP/JPY', tvSymbol: 'FX:GBPJPY', price: rates.JPY / rates.GBP })
   468→    }
   469→
   470→    for (const update of forexUpdates) {
   471→      try {
   472→        addSimulatedTick(update.internal, update.tvSymbol, update.price)
   473→      } catch (e) {
   474→        console.error(`[Forex] Error adding tick for ${update.internal}:`, e)
   475→      }
   476→    }
   477→
   478→    console.log(`[Forex] Updated ${forexUpdates.length} forex pairs`)
   479→  } catch (e) {
   480→    console.error(`[Forex] Poll error:`, e)
   481→  }
   482→}

   484→function addSimulatedTick(internal: string, tvSymbol: string, basePrice: number) {
   485→  const symbolData = symbolDataMap.get(internal)
   486→  if (!symbolData) return
   487→
   488→  // Update base price
   489→  symbolData.basePrice = basePrice
   490→
   491→  // Generate simulated tick with ±0.001% variation
   492→  const variation = (Math.random() - 0.5) * 0.0002 * basePrice
   493→  const price = parseFloat((basePrice + variation).toFixed(10))
   494→  const lastDigit = getLastDigit(price)
   495→  const timestamp = Date.now()
   496→
   497→  const tick: Tick = { price, timestamp, lastDigit }
   498→  symbolData.ticks.push(tick)
   499→  if (symbolData.ticks.length > MAX_TICKS) {
   500→    symbolData.ticks = symbolData.ticks.slice(-MAX_TICKS)
   501→  }
   502→
   503→  try {
   504→    const digits = computeDigitAnalysis(internal)
   505→    io.emit('tick', {
   506→      symbol: tvSymbol,
   507→      price,
   508→      lastDigit,
   509→      timestamp,
   510→      digits,
   511→    })
   512→  } catch (e) {
   513→    // Don't let socket.io errors crash the service
   514→  }
   515→}

   521→async function pollYahooPrices() {
   522→  const yahooSymbols = Object.entries(SYMBOL_MAP)
   523→    .filter(([_, m]) => m.source === 'yahoo' && m.yahooSymbol)
   524→    .map(([tv, m]) => ({ tv, internal: m.internal, yahoo: m.yahooSymbol! }))
   525→
   526→  // Poll in batches to avoid rate limiting
   527→  const batchSize = 5
   528→  for (let i = 0; i < yahooSymbols.length; i += batchSize) {
   529→    const batch = yahooSymbols.slice(i, i + batchSize)
   530→    const promises = batch.map(async (s) => {
   531→      try {
   532→        const url = `https://query1.finance.yahoo.com/v8/finance/chart/${s.yahoo}?interval=1d&range=1d`
   533→        const response = await fetch(url, {
   534→          headers: {
   535→            'User-Agent': 'Mozilla/5.0 (compatible; TickFeed/1.0)',
   536→          },
   537→        })
   538→        if (!response.ok) return null
   539→        const data = await response.json()
   540→        const result = data.chart?.result?.[0]
   541→        if (!result) return null
   542→
   543→        const meta = result.meta
   544→        const price = meta.regularMarketPrice
   545→        if (price === undefined || price === null) return null
   546→
   547→        return { ...s, price }
   548→      } catch {
   549→        return null
   550→      }
   551→    })
   552→
   553→    const results = await Promise.all(promises)
   554→
   555→    for (const result of results) {
   556→      if (!result) continue
   557→      try {
   558→        addSimulatedTick(result.internal, result.tv, result.price)
   559→      } catch (e) {
   560→        console.error(`[Yahoo] Error adding tick for ${result.internal}:`, e)
   561→      }
   562→    }
   563→
   564→    // Small delay between batches
   565→    if (i + batchSize < yahooSymbols.length) {
   566→      await new Promise(resolve => setTimeout(resolve, 500))
   567→    }
   568→  }
   569→
   570→  console.log(`[Yahoo] Polled ${yahooSymbols.length} stock/index symbols`)
   571→}
